32. Count the number of Fast stations.
SELECT VALUE COUNT(1) FROM c WHERE c.station_type = "Fast"
33. Count the number of stations in Pune.
SELECT VALUE COUNT(1) FROM c WHERE c.city = "Pune"

SELECT VALUE COUNT(1) FROM c WHERE c.revenue > 15000

Section H — GROUP BY
35. Count the number of stations in each city.
SELECT c.city, COUNT(1) AS TotalStations 
FROM c 
GROUP BY c.city
36. Count the number of stations by station_type .
SELECT c.station_type, COUNT(1) AS TotalStations 
FROM c 
GROUP BY c.station_type
37. Count the number of stations by status .
SELECT c.status, COUNT(1) AS TotalStations 
FROM c 
GROUP BY c.status
38. Find the total revenue for each city.
SELECT c.city, SUM(c.revenue) AS TotalRevenue 
FROM c 
GROUP BY c.city
39. Find the average revenue for each station type.
SELECT c.station_type, AVG(c.revenue) AS AvgRevenue 
FROM c 
GROUP BY c.station_type
40. Find the average uptime percentage for each city.
SELECT c.city, AVG(c.uptime_percent) AS AvgUptime 
FROM c 
GROUP BY c.city
41. Find the total number of daily sessions for each city.
SELECT c.city, SUM(c.daily_sessions) AS TotalSessions 
FROM c 
GROUP BY c.city
42. Find the average charging capacity for each station type.
SELECT c.station_type, AVG(c.capacity_kw) AS AvgCapacity 
FROM c 
GROUP BY c.station_type

Section J — Combined Queries
47. Find the top five stations based on revenue.
SELECT TOP 5 c.station_name, c.revenue 
FROM c 
ORDER BY c.revenue DESC
48. Find the top three stations based on daily sessions.
SELECT TOP 3 c.station_name, c.daily_sessions 
FROM c 
ORDER BY c.daily_sessions DESC
49. Display Active Fast stations ordered by revenue from highest to lowest.
SELECT * 
FROM c 
WHERE c.status = "Active" AND c.station_type = "Fast" 
ORDER BY c.revenue DESC
50. Display stations whose revenue is greater than 12000 and uptime is greater than 97 , ordered
by revenue from highest to lowest.
SELECT * 
FROM c 
WHERE c.revenue > 12000 AND c.uptime_percent > 97 
ORDER BY c.revenue DESC
51. Display Hyderabad or Bengaluru stations having more than 30 daily sessions.
SELECT * 
FROM c 
WHERE (c.city = "Hyderabad" OR c.city = "Bengaluru") AND c.daily_sessions > 30
52. Display Maintenance stations ordered by uptime percentage from lowest to highest.
SELECT * 
FROM c 
WHERE c.status = "Maintenance" 
ORDER BY c.uptime_percent ASC
53. Display station_name , city , revenue , and uptime_percent for stations whose
revenue is greater than 15000 .
SELECT c.station_name, c.city, c.revenue, c.uptime_percent 
FROM c 
WHERE c.revenue > 15000
54. Display Fast stations having a capacity of at least 120 , ordered by capacity from highest to
lowest.
SELECT * 
FROM c 
WHERE c.station_type = "Fast" AND c.capacity_kw >= 120 
ORDER BY c.capacity_kw DESC
