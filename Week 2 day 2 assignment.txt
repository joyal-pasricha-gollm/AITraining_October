import pandas as pd
import json
import numpy as np
import seaborn as sns
import sqlite3
import matplotlib.pyplot as plt
import counter


# 1. Load the JSON file using Python.
with open("semi_structured_data.json") as file:
    data = json.load(file)
print(data)
# 2. Print all incident IDs.
for incident in data:
    print(incident["incident_id"])
# 3. Print the city of each incident.
for incident in data:
    print(incident["location"]["city"])
# 4. Display incidents where severity is High .
for incident in data:
    if incident["severity"] == "High":
        print(incident)
# 5. Display incidents where category is Safety .
for incident in data:
    if incident["category"] == "Safety":
        print(incident)
# 6. Some incidents do not contain actions . Print:
#
# No actions recorded
# when the field is missing.
for incident in data:
    if "actions" in incident:
        print(f"{incident['incident_id']}: {incident['actions']}")
    else:
        print(f"{incident['incident_id']}: No actions recorded")
# 7. Count how many incidents belong to each severity level.
print("\n--- 7. Severity Level Counts ---")
severities = [incident["severity"] for incident in data]
counts = counter.Counter(severities)
for severity, count in counts.items():
    print(f"{severity}: {count}")

# 8. Add a new field: status = "Open" to every incident
for incident in data:
    incident["status"] = "Open"
print("Status assigned successfully.")

# 9. Save the updated records to: updated_incidents.json
with open("updated_incidents.json", "w") as file:
    json.dump(data, file, indent=4)
print("q9 Saved updated JSON to updated_incidents.json ")

# 10. Convert JSON into a Pandas DataFrame using pd.json_normalize()
df_json = pd.json_normalize(data)
columns_to_show = ["incident_id", "severity", "category", "location.city", "location.site"]
print(df_json[columns_to_show])
